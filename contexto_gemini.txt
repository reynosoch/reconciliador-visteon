function dinero(valor) {
return new Intl.NumberFormat("en-US", {
style: "currency",
currency: "USD",
maximumFractionDigits: 0,
}).format(valor);
}

function numero(valor) {
return new Intl.NumberFormat("en-US").format(Math.round(valor));
}

function App() {
const [datos] = useState(datosIniciales);

const [filtroArea, setFiltroArea] = useState("TODOS");

const [mostrarArchivos, setMostrarArchivos] = useState(false);

const datosCalculados = useMemo(() => {
return datos.map((item) => {
const fisicoPlanta = item.almacen + item.piso;
const fisicoTotal = fisicoPlanta + item.dsv;

const qadPlanta = item.qad - 0;

const varPlanta = fisicoPlanta - qadPlanta;
const deltaTotal = fisicoTotal - item.qad;

return {
...item,
fisicoPlanta,
fisicoTotal,
varPlanta,
varPlantaUsd: varPlanta * item.costo,
deltaTotal,
deltaTotalUsd: deltaTotal * item.costo,
};
});
}, [datos]);

const datosFiltrados = useMemo(() => {
let resultado = [...datosCalculados];

if (filtroArea !== "TODOS") {
resultado = resultado.filter(
(item) => item.responsable === filtroArea
);
}

if (busqueda.trim() !== "") {
resultado = resultado.filter((item) =>
item.pn.toLowerCase().includes(busqueda.toLowerCase())
);
}

if (filtroEspecial === "QAD_0") {
resultado = resultado.filter(
(item) => item.qad === 0 && item.fisicoTotal > 0
);
}

if (filtroEspecial === "FISICO_0") {
resultado = resultado.filter(
(item) => item.fisicoTotal === 0 && item.qad > 0
);
}

if (filtroEspecial === "PERDIDAS") {
resultado = resultado.filter((item) => item.deltaTotalUsd < 0);
}

if (filtroEspecial === "GANANCIAS") {
resultado = resultado.filter((item) => item.deltaTotalUsd > 0);
}

resultado.sort((a, b) => {
if (orden === "PERDIDA_USD") {
return a.deltaTotalUsd - b.deltaTotalUsd;
}

if (orden === "GANANCIA_USD") {
return b.deltaTotalUsd - a.deltaTotalUsd;
}

if (orden === "PERDIDA_PZAS") {
return a.deltaTotal - b.deltaTotal;
}

if (orden === "GANANCIA_PZAS") {
return b.deltaTotal - a.deltaTotal;
}

return 0;
});

return resultado;
}, [
datosCalculados,
filtroArea,
filtroEspecial,
busqueda,
orden,
]);

.reduce((total, item) => total + item.deltaTotalUsd, 0);

.filter((item) => item.deltaTotalUsd > 0)
.reduce((total, item) => total + item.deltaTotalUsd, 0);

return {

fisicoDsv,
fisicoTotal,
qadTotal,
varPlanta,
deltaTotal,
perdida,
ganancia,
};

import React, { useMemo, useState } from "react";
import Header from "./components/Header";
import MetricCard from "./components/MetricCard";
import Hallazgos from "./components/Hallazgos";
import Filters from "./components/Filters";
import InventoryCharts from "./components/InventoryCharts";
import InventoryTable from "./components/InventoryTable";
import {
 inventarioMock,
 cortesInventario,
 archivosFuente,
} from "./data/mockData";
function App() {
 const [filtro, setFiltro] = useState("TODOS");
 const [orden, setOrden] = useState("PERDIDA_USD");
 const [busqueda, setBusqueda] = useState("");
 // ============================================================
 // ÚLTIMO CORTE
 // ============================================================
 const ultimoCorte =
   cortesInventario[cortesInventario.length - 1];
 // ============================================================
 // FILTROS Y ORDEN
 // ============================================================
 const inventarioFiltrado = useMemo(() => {
   let datos = [...inventarioMock];
   if (filtro === "ALMACÉN") {
     datos = datos.filter((item) => item.almacen > 0);
   }
   if (filtro === "PISO") {
     datos = datos.filter((item) => item.piso > 0);
   }
   if (filtro === "CONSIGNACIÓN") {
     datos = datos.filter((item) => item.dsv > 0);
   }
   if (busqueda.trim() !== "") {
     datos = datos.filter((item) =>
item.pn
         .toLowerCase()
         .includes(busqueda.toLowerCase())
     );
   }
   if (orden === "PERDIDA_USD") {
     datos.sort((a, b) => a.varUsd - b.varUsd);
   }
   if (orden === "GANANCIA_USD") {
     datos.sort((a, b) => b.varUsd - a.varUsd);
   }
   if (orden === "PERDIDA_PZAS") {
     datos.sort((a, b) => a.varPlanta - b.varPlanta);
   }
   if (orden === "GANANCIA_PZAS") {
     datos.sort((a, b) => b.varPlanta - a.varPlanta);
   }
   return datos;

 const metricas = useMemo(() => {
   const total4Wall = inventarioMock.reduce(
     (total, item) =>
       total + item.almacen + item.piso,
     0
   );
   const totalDsv = inventarioMock.reduce(
     (total, item) => total + item.dsv,
     0
   );
   const totalFisico =
     total4Wall + totalDsv;
   const totalQad = inventarioMock.reduce(
     (total, item) => total + item.qad,
     0
   );
   const varPlanta = inventarioMock.reduce(
     (total, item) => total + item.varPlanta,
     0
   );
   const impactoUsd = inventarioMock.reduce(
     (total, item) => total + item.varUsd,
     0
   );
   const pnsConDiferencia =
     inventarioMock.filter(
       (item) => item.varPlanta !== 0
     ).length;
   return {
     total4Wall,
     totalDsv,
     totalFisico,
     totalQad,
     varPlanta,
     impactoUsd,
     pnsConDiferencia,
   };
 }, []);
 // ============================================================
 // IMPACTO POR UBICACIÓN
 // ============================================================
 const impactoUbicacion = useMemo(() => {
   const resultado = {
     almacen: 0,
     piso: 0,
     dsv: 0,
   };
   inventarioMock.forEach((item) => {
     resultado.almacen +=
       item.varAlmacen * item.costo;
     resultado.piso +=
       item.varPiso * item.costo;
     resultado.dsv +=
       item.varDsv * item.costo;
   });
   return resultado;
 }, []);
 // ============================================================

     );
   const fisicoCeroQad =
     inventarioMock.filter(

         item.almacen + item.piso === 0 &&
         item.qad > 0
     );
   const perdidaEconomica =
     [...inventarioMock]
       .filter((item) => item.varUsd < 0)
       .sort(

       )
       .slice(0, 5);

       .sort(
         (a, b) => b.varUsd - a.varUsd
       )
       .slice(0, 5);
   return {

     fisicoCeroQad,
     perdidaEconomica,
